.AnyAsync(u => u.Id == userId && u.TenantId == tenantId, ct);
if (!userExists)
return NotFound(new
{
success = false,
errorCode = "NOT_FOUND",
message = "User not found in this tenant."
});
var alreadyAssigned = await _iamDb.UserStoreAssignments
.AnyAsync(a =>
a.UserId == userId &&
a.StoreId == request.StoreId &&
a.TenantId == tenantId, ct);
if (!alreadyAssigned)

_iamDb.UserStoreAssignments.Add(new UserStoreAssignment

UserId = userId,
